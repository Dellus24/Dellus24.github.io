# Retired scratch experiment, still served. Not content. (HYG-05)
Disallow: /pages/

# Content SOURCES, not pages. js/config.js is generated from these, so every
# line already appears on the site proper — indexing them would only create
# plain-text duplicates competing with the real pages. content/cv.txt also
# carries the email and date of birth that cv.html is noindex for, and a .txt
# gets indexed far more readily than the .js it is compiled into.
# Remove these two lines if you ever want the raw sources crawlable.
Disallow: /content/
Disallow: /*project.txt$

# NOTE: /cv.html is deliberately NOT disallowed here.
# It carries <meta name="robots" content="noindex, nofollow">, and a crawler
# has to be allowed to fetch the page in order to read that tag. Blocking it
# here instead would let the bare URL still surface in results (linked from
# elsewhere, no snippet) with the noindex never seen.
